    <section id="riesgos" class="tarjeta">
      <h2 class="titulo-destacado">02. Riesgos digitales</h2>
      <p>
        Pasar demasiado tiempo utilizando dispositivos sin descansos
        ni límites puede afectar nuestras actividades cotidianas.
        Reconocer los riesgos nos ayuda a prevenir problemas.
      </p>

      <img
        class="imagen-salud"
        src="https://images.unsplash.com/photo-1511707171634-5f897ff02aa9?auto=format&fit=crop&w=1000&q=85"
        alt="Teléfono celular utilizado para actividades digitales"
        loading="lazy"
      >

      <h3 class="subtitulo">Principales consecuencias</h3>
      <ul>
        <li><strong>Falta de concentración:</strong> las notificaciones pueden interrumpir el estudio.</li>
        <li><strong>Alteraciones del sueño:</strong> utilizar pantallas hasta tarde puede dificultar el descanso.</li>
        <li><strong>Fatiga visual:</strong> mirar de cerca una pantalla durante mucho tiempo puede causar cansancio ocular.</li>
        <li><strong>Molestias físicas:</strong> una postura inadecuada puede causar molestias en el cuello y la espalda.</li>
        <li><strong>Menor actividad física:</strong> permanecer sentado durante periodos prolongados reduce las oportunidades de movimiento.</li>
        <li><strong>Riesgos de privacidad:</strong> compartir información personal puede poner en peligro nuestra seguridad.</li>
      </ul>

      <div class="destacado">
        <strong>Recuerda:</strong> la tecnología no es el problema por sí misma.
        Lo importante es cómo, cuándo y para qué la utilizamos.
      </div>

        <p>
          El ciberacoso (o ciberbullying) es uno de los riesgos más delicados en internet. 
          Ocurre cuando se usa la tecnología para intimidar, amenazar o burlarse de alguien de forma reiterada.
        </p>
        <p><strong>¿Qué hacer si ocurre?</strong></p>
        <ul>
          <li>No respondas a las provocaciones ni alimentes el conflicto.</li>
          <li>Guarda pruebas (haz capturas de pantalla).</li>
          <li>Bloquea al usuario de inmediato.</li>
          <li>Comunícaselo a tus padres, maestros o una persona de confianza.</li>
        </ul>

      <ul>

        <li>Aplicar la regla 20-20-20 para descansar la vista.</li>
        <li>Mantener la espalda apoyada y el dispositivo a una altura cómoda.</li>
        <li>Realizar actividades al aire libre y convivir en persona.</li>
      </ul>

      <div class="destacado">

        cada 20 minutos, mira durante 20 segundos un objeto situado

        reducir la fatiga visual.

    // Barra de progreso de lectura y botón Volver a la Bienvenida
    const btnArriba = document.getElementById('btn-arriba');

    window.addEventListener('scroll', () => {
      const winScroll = document.documentElement.scrollTop || document.body.scrollTop;
      const height = document.documentElement.scrollHeight - document.documentElement.clientHeight;
      const scrolled = (winScroll / height) * 100;
      document.getElementById('barra-lectura').style.width = scrolled + '%';

      // Mostrar/Ocultar botón al bajar más de 300px
      if (winScroll > 300) {
        btnArriba.classList.add('mostrar');
      } else {
        btnArriba.classList.remove('mostrar');
      }
    });

    // Desplegable de recursos
    const dropdown = document.getElementById('menu-recursos');
    const btnDropdown = document.getElementById('btn-dropdown');

    btnDropdown.addEventListener('click', (e) => {
      e.stopPropagation();
      const activo = dropdown.classList.toggle('activo');
      btnDropdown.setAttribute('aria-expanded', activo);
    });

    document.addEventListener('click', () => {
      if (dropdown.classList.contains('activo')) {
        dropdown.classList.remove('activo');
        btnDropdown.setAttribute('aria-expanded', 'false');
      }
    });

    // Animación de aparición de tarjetas al hacer scroll
    const observer = new IntersectionObserver((entries) => {
      entries.forEach(entry => {
        if (entry.isIntersecting) {
          entry.target.classList.add('visible');
        }
      });
    }, { threshold: 0.1 });

    document.querySelectorAll('.tarjeta').forEach(tarjeta => {
      observer.observe(tarjeta);
    });

    // Lógica básica interactiva del reproductor de música
    const btnPlayPause = document.getElementById('btn-play-pause');
    const estadoReproduccion = document.getElementById('estado-reproduccion');
    let reproduciendo = false;

    btnPlayPause.addEventListener('click', () => {
      reproduciendo = !reproduciendo;
      if (reproduciendo) {
        btnPlayPause.textContent = '⏸';
        estadoReproduccion.textContent = 'Reproduciendo...';
      } else {
        btnPlayPause.textContent = '▶';
        estadoReproduccion.textContent = 'Pausado';
      }
    });
